%% Afin de supprimer les symbols de navigation Beamer

\section{Le thème \emph{ecl24}}

\begin{frame}[fragile]{\insertsection} % 'fragile' nécessaire à verbatim
  %% Bloc important
  \begin{alertblock}{Prérequis}
    Pour utiliser le thème Beamer \emph{ecl24}, il vous faut appeler les modules \emph{tikz}, \emph{ifthen} et \emph{etoolbox}.

    Pensez à vérifier que les images utilisée par le thème Beamer \emph{ecl24} sont accessibles en indiquant par exemple le chemin vers ces images.
  \end{alertblock}

  %% Bloc d'exemple
  \begin{exampleblock}{Utilisation du thème \emph{ecl24}}
    Ces commandes sont à insérer dans l'en-tête de votre fichier latex (\emph{ie} avant \verb+\begin{document}+) :
    \begin{verbatim}
      \usepackage{tikz}     
      \usepackage{ifthen,etoolbox}
      \graphicspath{{IMAGES/}}
      \usetheme{ecl24}
    \end{verbatim}
  \end{exampleblock}
\end{frame}


\subsection{Licence creative commons BY-NC-ND}

\begin{frame}[fragile]{\insertsection}{\insertsubsection}
  %% Bloc normal
  \begin{block}{Utiliser la licence creative commons BY-NC-ND}
    Pour utiliser la licence creative commons BY-NC-ND, activez l'option \emph{ccbyncnd} en appelant le thème \emph{ecl24} :

    \verb+\usetheme[ccbyncnd]{ecl24}+
  \end{block}
\end{frame}

\subsection{Image de fond de la page de garde avec photo}

\begin{frame}[fragile]{\insertsection}{\insertsubsection}
  %% Bloc normal
  \begin{block}{Utiliser l'image de fond dans la page de garde}
    Pour utiliser que la page de garde soit celle avec une photo, activez l'option \emph{p1photo} en appelant le thème \emph{ecl24} :

    \verb+\usetheme[p1photo]{ecl24}+

    Avec en plus la licence creative commons BY-NC-ND :

    \verb+\usetheme[ccbyncnd,p1photo]{ecl24}+
\end{block}
\end{frame}

\subsection{Dernière page dans le thème défini par l'école centrale de Lyon}

\begin{frame}[fragile]{\insertsection}{\insertsubsection}
  %% Bloc normal
  \begin{block}{Utiliser le thème de l'école pour la dernière page}
    Pour utiliser le thème de l'école centrale de Lyon dans la dernière, activez l'option \emph{finecl} en appelant le thème \emph{ecl24} :\\
    \verb+\usetheme[finecl]{ecl24}+

    Avec en plus la licence creative commons BY-NC-ND :\\
    \verb+\usetheme[ccbyncnd,finecl]{ecl24}+

    N'oubliez pas d'ajouter une page vide en fin de présentation :\\
    \verb+\begin{frame}+\\
    \verb+\end{frame}+
  \end{block}
\end{frame}
